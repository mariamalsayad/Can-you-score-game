cmake_minimum_required(VERSION 3.16)
project(CanYouScore VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_GAME "Build the graphical game" ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(physics_tests tests/physics_tests.cpp)
  target_include_directories(physics_tests PRIVATE src)
  add_test(NAME physics COMMAND physics_tests)
endif()
if(BUILD_GAME)
  find_package(raylib 5.5 QUIET)
  if(NOT raylib_FOUND)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
    add_subdirectory(vendor/raylib)
  endif()
  add_executable(can_you_score src/main.cpp)
  target_link_libraries(can_you_score PRIVATE raylib)
  if(MSVC)
    target_compile_options(can_you_score PRIVATE /W4)
  else()
    target_compile_options(can_you_score PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
